cmake_minimum_required(VERSION 3.22)
project(GrainBrain VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(GrainBrain COMPANY_NAME "GrainBrain" BUNDLE_ID "audio.grainbrain.synth" PLUGIN_MANUFACTURER_CODE GrBr PLUGIN_CODE GrB1 FORMATS AU VST3 Standalone PRODUCT_NAME "GrainBrain" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(GrainBrain)
target_sources(GrainBrain PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(GrainBrain PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(GrainBrain PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_audio_formats juce::juce_gui_extra juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
